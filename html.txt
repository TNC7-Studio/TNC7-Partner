    <script>
        const terminal = document.getElementById('aiLog');
        let outputMode = 'image';
        let excelBlobUrl = null;
        let currentExcelBlob = null; 
        let outputDirHandle = null;  

        async function selectOutputFolder() {
            try {
                if (!window.showDirectoryPicker) return null;
                
                // Mencegah SecurityError jika dijalankan di dalam iFrame (Preview Environment)
                if (window.self !== window.top) {
                    console.warn("Folder picker API tidak diizinkan di dalam iFrame (Preview). Beralih ke unduhan biasa.");
                    return null;
                }

                outputDirHandle = await window.showDirectoryPicker({ mode: 'readwrite' });
                log(`> Folder Auto-Save diatur ke: <span class="text-cyan-400 font-bold">[ ${outputDirHandle.name} ]</span>`);
                return outputDirHandle;
            } catch (err) {
                // Menyembunyikan log error merah, mengubahnya menjadi peringatan biasa
                if (err.name !== 'AbortError') {
                    console.warn("Folder picker batal/gagal. Beralih ke unduhan normal.");
                }
                return null;
            }
        }

        async function saveFileHelper(blob, filename) {
            if (outputDirHandle) {
                try {
                    const fileHandle = await outputDirHandle.getFileHandle(filename, { create: true });
                    const writable = await fileHandle.createWritable();
                    await writable.write(blob);
                    await writable.close();
                    return true;
                } catch (err) {
                    console.error("Gagal save ke folder, dialihkan ke normal:", err);
                }
            }
            const link = document.createElement('a');
            const url = URL.createObjectURL(blob);
            link.href = url;
            link.download = filename;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            setTimeout(() => URL.revokeObjectURL(url), 100);
            return true;
        }

        function renderEmptyGrids() {
            const buildGrid = (tableId) => {
                const table = document.getElementById(tableId);
                if(!table) return;
                table.innerHTML = ''; 
                let tbody = document.createElement('tbody');
                for (let r = 1; r <= 35; r++) {
                    let tr = document.createElement('tr');
                    tr.className = "bg-[#0a0a0a]";
                    for (let c = 1; c <= 20; c++) {
                        let td = document.createElement('td');
                        td.className = "h-[21px] min-w-[64px]";
                        tr.appendChild(td);
                    }
                    tbody.appendChild(tr);
                }
                table.appendChild(tbody);
            };
            buildGrid('emptyGridOutput');
        }

        window.onload = function() {
            renderEmptyGrids();
        };

        const getSafeValIn = (cell) => {
            if (!cell || cell.value === null) return '';
            if (typeof cell.value === 'object') {
                if (cell.value.result !== undefined) return cell.value.result;
                if (cell.value.richText) return cell.value.richText.map(rt => rt.text).join('');
                if (cell.value instanceof Date) return cell.value.toLocaleDateString('id-ID');
                return '';
            }
            return cell.value;
        };

        const renderExactExcelTable = (sheet, tableElementId) => {
            const table = document.getElementById(tableElementId);
            table.innerHTML = '';
            
            table.className = "bg-white text-left text-black border-collapse";
            table.style.width = "max-content"; 
            table.style.tableLayout = "fixed"; 
            table.style.backgroundColor = "#ffffff";

            const maxRows = Math.max(sheet.rowCount, 10);
            const maxCols = Math.max(sheet.columnCount, 10);

            let mergeMap = {};
            if (sheet.model.merges) {
                sheet.model.merges.forEach(m => {
                    const [start, end] = m.split(':');
                    const startCell = sheet.getCell(start);
                    const endCell = sheet.getCell(end);
                    const r1 = startCell.row, c1 = startCell.col;
                    const r2 = endCell.row, c2 = endCell.col;

                    mergeMap[`${r1}-${c1}`] = { rowSpan: r2 - r1 + 1, colSpan: c2 - c1 + 1, isMaster: true };
                    for (let r = r1; r <= r2; r++) {
                        for (let c = c1; c <= c2; c++) {
                            if (r !== r1 || c !== c1) mergeMap[`${r}-${c}`] = { isMaster: false };
                        }
                    }
                });
            }

            const getColWidthPx = (colIndex) => {
                let col = sheet.getColumn(colIndex);
                let w = col.width || 10;
                return Math.floor(w * 22); 
            };

            const getBStyle = (b) => {
                if (!b) return null;
                let c = '#000000';
                if (b.color && b.color.argb && b.color.argb !== '00000000') c = '#' + b.color.argb.slice(-6);
                let w = '3px'; 
                if (b.style === 'medium') w = '4px';
                if (b.style === 'thick') w = '5px';
                if (b.style === 'hair' || b.style === 'dotted') return `2px dotted ${c}`;
                if (b.style === 'dashed') return `2px dashed ${c}`;
                return `${w} solid ${c}`;
            };

            let tbody = document.createElement('tbody');
            for (let r = 1; r <= maxRows; r++) {
                let row = sheet.getRow(r);
                let tr = document.createElement('tr');
                let rowHeight = row ? (row.height || 18) : 18;
                let hPx = Math.floor(rowHeight * 3.0); 
                tr.style.height = hPx + 'px';

                for (let c = 1; c <= maxCols; c++) {
                    let mergeInfo = mergeMap[`${r}-${c}`];
                    if (mergeInfo && !mergeInfo.isMaster) continue;

                    let td = document.createElement('td');
                    let wPx = getColWidthPx(c);
                    if (mergeInfo && mergeInfo.isMaster) {
                        wPx = 0;
                        for (let mc = 0; mc < mergeInfo.colSpan; mc++) wPx += getColWidthPx(c + mc);
                    }

                    td.style.width = wPx + 'px';
                    td.style.minWidth = wPx + 'px';
                    td.style.maxWidth = "none"; 
                    td.style.boxSizing = "border-box";
                    td.style.padding = "18px 28px"; 
                    td.style.overflow = "visible"; 
                    td.style.whiteSpace = "nowrap";

                    td.style.fontFamily = "'Arial Black', 'Segoe UI Black', Arial, sans-serif"; 
                    td.style.fontSize = "28px"; 
                    td.style.fontWeight = "800"; 
                    td.style.color = "#000000"; 
                    td.style.border = "none"; 
                    td.style.letterSpacing = "0.03em";

                    if (mergeInfo && mergeInfo.isMaster) {
                        if (mergeInfo.rowSpan > 1) td.rowSpan = mergeInfo.rowSpan;
                        if (mergeInfo.colSpan > 1) td.colSpan = mergeInfo.colSpan;
                    }

                    let cell = row ? row.getCell(c) : null;
                    let val = cell ? getSafeValIn(cell) : '';

                    if (typeof val === 'number') {
                        td.innerText = val.toLocaleString('id-ID', { minimumFractionDigits: 0, maximumFractionDigits: 0 });
                        td.style.textAlign = 'right';
                    } else if (typeof val === 'string') {
                        // Check if the string is a formatted number with decimals
                        let match = val.match(/^[0-9.,]+$/);
                        if (match && val.includes(',')) {
                            let parts = val.split(',');
                            if (parts.length === 2 && parts[1].length <= 2) {
                                td.innerText = parts[0];
                            } else {
                                td.innerText = val;
                            }
                            td.style.textAlign = 'right';
                        } else {
                            td.innerText = val;
                        }
                    } else {
                        td.innerText = val;
                    }

                    if (val === '') td.innerHTML = '&nbsp;';

                    if (cell) {
                        let isMainTitle = false;
                        let isHeaderInfo = false;
                        let strVal = String(val).toUpperCase();
                        
                        // Deteksi Judul Utama
                        if (strVal.includes('REKAP') || strVal.includes('YOUTUBE') || strVal.includes('PENGHASILAN') || strVal.includes('HASIL')) {
                            isMainTitle = true;
                        } else if (r <= 4 && mergeInfo && mergeInfo.isMaster) {
                            isMainTitle = true; 
                        } else if (r <= 4) {
                            // Deteksi area Header (Partner, Nama Channel, dll)
                            isHeaderInfo = true;
                        }

                        if (cell.font) {
                            if (cell.font.bold) td.style.fontWeight = '900'; 
                            if (cell.font.italic) td.style.fontStyle = 'italic';
                            
                            if (cell.font.size) {
                                let extraSize = 16; // Ukuran normal tabel
                                if (isMainTitle) {
                                    extraSize = 42; // Memperbesar Ekstra Judul Utama
                                } else if (isHeaderInfo) {
                                    extraSize = 28; // Memperbesar Info Partner
                                }
                                td.style.fontSize = (cell.font.size + extraSize) + 'pt'; 
                            }
                            
                            if (cell.font.color && cell.font.color.argb && cell.font.color.argb !== '00000000') {
                                td.style.color = '#' + cell.font.color.argb.slice(-6);
                            }
                        } else {
                            if (isMainTitle) {
                                td.style.fontSize = '50pt'; 
                                td.style.fontWeight = '900';
                            } else if (isHeaderInfo) {
                                td.style.fontSize = '34pt'; 
                                td.style.fontWeight = '900';
                            }
                        }

                        if (cell.fill && cell.fill.type === 'pattern' && cell.fill.fgColor && cell.fill.fgColor.argb) {
                            if (cell.fill.fgColor.argb !== '00000000') {
                                td.style.backgroundColor = '#' + cell.fill.fgColor.argb.slice(-6);
                            }
                        }
                        if (cell.alignment) {
                            if (cell.alignment.horizontal) td.style.textAlign = cell.alignment.horizontal;
                            if (cell.alignment.vertical) {
                                let va = cell.alignment.vertical;
                                if(va === 'center') va = 'middle';
                                td.style.verticalAlign = va;
                            }
                        }
                        
                        let borderTop, borderBottom, borderLeft, borderRight;
                        if (cell.border) {
                            borderTop = cell.border.top;
                            borderBottom = cell.border.bottom;
                            borderLeft = cell.border.left;
                            borderRight = cell.border.right;
                        }

                        if (mergeInfo && mergeInfo.isMaster) {
                            let rightCol = c + (mergeInfo.colSpan || 1) - 1;
                            let bottomRow = r + (mergeInfo.rowSpan || 1) - 1;
                            
                            if (rightCol > c) {
                                let rightCell = row ? row.getCell(rightCol) : null;
                                if (rightCell && rightCell.border && rightCell.border.right) borderRight = rightCell.border.right;
                            }
                            if (bottomRow > r) {
                                let bottomRowObj = sheet.getRow(bottomRow);
                                let bottomCell = bottomRowObj ? bottomRowObj.getCell(c) : null;
                                if (bottomCell && bottomCell.border && bottomCell.border.bottom) borderBottom = bottomCell.border.bottom;
                            }
                            if (bottomRow > r && rightCol > c) {
                                let brRowObj = sheet.getRow(bottomRow);
                                let brCell = brRowObj ? brRowObj.getCell(rightCol) : null;
                                if (brCell && brCell.border) {
                                    if (!borderRight && brCell.border.right) borderRight = brCell.border.right;
                                    if (!borderBottom && brCell.border.bottom) borderBottom = brCell.border.bottom;
                                }
                            }
                        }

                        if (borderTop) td.style.borderTop = getBStyle(borderTop);
                        if (borderBottom) td.style.borderBottom = getBStyle(borderBottom);
                        if (borderLeft) td.style.borderLeft = getBStyle(borderLeft);
                        if (borderRight) td.style.borderRight = getBStyle(borderRight);
                    }

                    tr.appendChild(td);
                }
                tbody.appendChild(tr);
            }
            table.appendChild(tbody);
        };

                        ws.eachRow((row, rowNumber) => {
                            row.eachCell((cell, colNumber) => {
                                let valText = '';
                                if (cell.value) {
                                    if (typeof cell.value === 'string') valText = cell.value.toUpperCase();
                                    else if (cell.value.richText) valText = cell.value.richText.map(rt => rt.text).join('').toUpperCase();

                            if (valText.includes('70%')) col70 = colNumber;

                            if (col100 !== -1 && col70 !== -1 && col30 !== -1 && headerRowNumber === -1) {
                                headerRowNumber = rowNumber;
                                targetSheet = ws;
                            }
                        });
                        if (targetSheet) return;
                    });
                    if (targetSheet) break;
                }

                if (targetSheet && col100 !== -1 && col70 !== -1 && col30 !== -1) {
                    let headerRow = targetSheet.getRow(headerRowNumber);
                    
                    let cell70 = headerRow.getCell(col70);
                    if (typeof cell70.value === 'string') cell70.value = cell70.value.replace(/70%/g, '80%');
                    else if (cell70.value && cell70.value.richText) cell70.value.richText.forEach(rt => rt.text = rt.text.replace(/70%/g, '80%'));

                    let cell30 = headerRow.getCell(col30);
                    if (typeof cell30.value === 'string') cell30.value = cell30.value.replace(/30%/g, '20%');
                    else if (cell30.value && cell30.value.richText) cell30.value.richText.forEach(rt => rt.text = rt.text.replace(/30%/g, '20%'));

                    targetSheet.eachRow((row, rowNumber) => {
                        if (rowNumber > headerRowNumber) {
                            const cell100 = row.getCell(col100);
                            let val100 = 0;
                            if (cell100.value !== null) {
                                if (typeof cell100.value === 'number') {
                                    val100 = cell100.value;
                                } else if (typeof cell100.value === 'object' && cell100.value.result !== undefined) {
                                    val100 = Number(cell100.value.result);
                                } else {
                                    let strVal = cell100.value.toString();
                                    val100 = parseFloat(strVal.replace(/[^0-9,-]+/g,"").replace(',', '.'));
                                }
                            }

                            if (!isNaN(val100) && val100 > 0) {
                                let val80 = Math.round(val100 * 0.8);
                                let val20 = val100 - val80;
                                row.getCell(col70).value = val80;
                                row.getCell(col30).value = val20;
                            }
                        }
                    });

                    outputMode = 'excel';
                    
                    let baseFilename = file.name.replace(/\.[^/.]+$/, "");
                    baseFilename = baseFilename.replace(/Report\s*YT|Report_YT/gi, '').replace(/[()]/g, '').replace(/_/g, ' ').replace(/\s+/g, ' ').trim();
                    
                    // Filter bulan dan tahun untuk UI Input (berlaku untuk JPG & XLSX)
                    const monthRegex = /[_\s]?(januari|februari|maret|april|mei|juni|juli|agustus|september|oktober|november|desember|january|february|march|may|june|july|august|october|december)[_\s\d]*$/i;
                    baseFilename = baseFilename.replace(monthRegex, '');
                    baseFilename = baseFilename.replace(/[_\s][0-9]{2,4}$/, '').trim();
                    
                    let defaultFilename = (baseFilename || 'TNC7 Partner');
                    
                    document.getElementById('filenameInput').value = defaultFilename;

                    const outputBuffer = await workbook.xlsx.writeBuffer();
                    currentExcelBlob = new Blob([outputBuffer], { type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' });
                    
                    if (excelBlobUrl) URL.revokeObjectURL(excelBlobUrl);
                    excelBlobUrl = URL.createObjectURL(currentExcelBlob);

                    document.getElementById('scanLine').style.display = 'none';
                    document.getElementById('outputPrompt').classList.add('hidden');
                    document.getElementById('dropZone').classList.add('hidden');
                    if (document.getElementById('excelOutputWrapper')) {
                        document.getElementById('excelOutputWrapper').classList.remove('hidden');
                    }
                    if (document.getElementById('excelOutput')) {
                        document.getElementById('excelOutput').classList.remove('hidden');
                    }

                    document.getElementById('btnDownloadXlsx').classList.remove('hidden');

        async function runTesseractOCR(imgElement) {
            log("Engine OCR Aktif: Memindai matriks dokumen (Resolusi Tinggi)...");
            
            const scaleFactor = 2; 
            const upCanvas = document.createElement('canvas');
            upCanvas.width = imgElement.width * scaleFactor;
            upCanvas.height = imgElement.height * scaleFactor;
            const upCtx = upCanvas.getContext('2d');
            upCtx.imageSmoothingEnabled = false; 
            upCtx.drawImage(imgElement, 0, 0, upCanvas.width, upCanvas.height);
            
            try {
                const worker = await Tesseract.createWorker('eng');
                await worker.setParameters({
                    tessedit_pageseg_mode: Tesseract.PSM.SPARSE_TEXT, 
                });

                const result = await worker.recognize(upCanvas);
                await worker.terminate();

                let words = result.data.words.map(w => ({
                    text: w.text.trim(),
                    bbox: {
                        x0: w.bbox.x0 / scaleFactor, y0: w.bbox.y0 / scaleFactor,
                        x1: w.bbox.x1 / scaleFactor, y1: w.bbox.y1 / scaleFactor
                    },
                    centerX: (w.bbox.x0 + w.bbox.x1) / (2 * scaleFactor),
                    centerY: (w.bbox.y0 + w.bbox.y1) / (2 * scaleFactor)
                }));

                processPixelPerfect(words, imgElement);

            } catch (error) {
                console.error(error);
                log("<span class='text-red-500'>Error: Gagal memproses gambar.</span>");
                document.getElementById('scanLine').style.display = 'none';
            }
        }

        function processPixelPerfect(words, imgElement) {
            log("> Menganalisis Zona Fokus: (IDR), (70%), dan (30%)...");

            let defaultFilename = "TNC7 Partner";
            let partnerWords = words.filter(w => w.text.trim().toLowerCase() === 'partner');
            partnerWords.sort((a, b) => a.centerY - b.centerY); 
            let partnerLabel = partnerWords.length > 0 ? partnerWords[0] : null;

            if (partnerLabel) {
                let nameWords = words.filter(w => 
                    Math.abs(w.centerY - partnerLabel.centerY) < 15 && 
                    w.centerX > partnerLabel.centerX + 30
                );

                if (nameWords.length > 0) {
                    nameWords.sort((a, b) => a.centerX - b.centerX);
                    let rawName = nameWords.map(w => w.text).join(' ');
                    let cleanName = rawName.replace(/[()]/g, '').replace(/[^a-zA-Z0-9\s]/g, '').trim();
                    
                    if (cleanName.length > 0) {
                        defaultFilename = cleanName;
                        defaultFilename = defaultFilename.replace(/Report\s*YT|Report_YT/gi, '').replace(/[()]/g, '').replace(/_/g, ' ').replace(/\s+/g, ' ').trim();
                        
                        // Filter bulan dan tahun untuk hasil scan OCR gambar
                        const monthRegex = /[_\s]?(januari|februari|maret|april|mei|juni|juli|agustus|september|oktober|november|desember|january|february|march|may|june|july|august|october|december)[_\s\d]*$/i;
                        defaultFilename = defaultFilename.replace(monthRegex, '');
                        defaultFilename = defaultFilename.replace(/[_\s][0-9]{2,4}$/, '').trim();
                        
                        if(!defaultFilename) defaultFilename = "TNC7 Partner";
                    }
                }
            }
            
            document.getElementById('filenameInput').value = defaultFilename;

            const canvas2 = document.getElementById('canvas2');
            const ctx2 = canvas2.getContext('2d', { willReadFrequently: true });
            canvas2.width = imgElement.width;
            canvas2.height = imgElement.height;
            ctx2.drawImage(imgElement, 0, 0); 
            const imgData = ctx2.getImageData(0, 0, canvas2.width, canvas2.height);

            let headerIDR = null, header70 = null, header30 = null;
            words.forEach(w => {
                let t = w.text.toUpperCase();
                if (t.includes('IDR')) headerIDR = w;
                if (t.includes('70%') || t.includes('(70%)')) header70 = w;
                if (t.includes('30%') || t.includes('(30%)')) header30 = w;
            });

            if (!headerIDR || !header70 || !header30) {
                log("<span class='text-red-400'>Error: Kolom (IDR) / (70%) / (30%) tidak terdeteksi jelas.</span>");
                document.getElementById('scanLine').style.display = 'none';
                return;
            }

            function patchText(targetWords, newText, isHeader = false) {
                if (!targetWords || targetWords.length === 0) return;
                let minX = Math.min(...targetWords.map(w => w.bbox.x0));
                let maxX = Math.max(...targetWords.map(w => w.bbox.x1));
                let minY = Math.min(...targetWords.map(w => w.bbox.y0));
                let maxY = Math.max(...targetWords.map(w => w.bbox.y1));

                let padX = 1; 
                let padY = 2;
                let sampleX = Math.floor(minX + 2); 
                let sampleY = Math.floor(minY - 2); 
                let idx = (sampleY * imgData.width + sampleX) * 4;
                let r = imgData.data[idx], g = imgData.data[idx+1], b = imgData.data[idx+2];
                let bgColor = `rgb(${r}, ${g}, ${b})`;

                if (r > 230 && g > 230 && b > 230) {
                    bgColor = "#FFFFFF";
                }

                ctx2.fillStyle = bgColor;
                ctx2.fillRect(minX - padX, minY - padY, (maxX - minX) + (padX * 2), (maxY - minY) + (padY * 2));

                let height = maxY - minY;
                let fontSize = Math.max(20, height * 1.6); 
                ctx2.fillStyle = "#000000";
                
                if (isHeader) {
                    ctx2.font = `700 ${fontSize}px Calibri, Arial, sans-serif`; 
                    ctx2.textAlign = "center";
                    ctx2.textBaseline = "middle";
                    let centerX = (minX + maxX) / 2;
                    ctx2.fillText(newText, centerX, (minY + maxY) / 2 + 1);
                } else {
                    ctx2.font = `900 ${fontSize}px Calibri, Arial, sans-serif`;
                    ctx2.textAlign = "right";
                    ctx2.textBaseline = "middle";
                    let textX = maxX - 1;
                    let textY = (minY + maxY) / 2 + 1;
                    ctx2.fillText(newText, textX, textY); 
                }
            }

            patchText([header70], "(80%)", true);
            patchText([header30], "(20%)", true);

            let idrNumbers = [];
            words.forEach(w => {
                let val = cleanNumber(w.text);
                if (val > 0 && Math.abs(w.centerX - headerIDR.centerX) < 60 && w.centerY > headerIDR.centerY) {
                    let existing = idrNumbers.find(n => Math.abs(n.centerY - w.centerY) < 10);
                    if (existing) {
                        existing.text += w.text;
                        existing.value = cleanNumber(existing.text);
                        existing.bbox.x1 = Math.max(existing.bbox.x1, w.bbox.x1);
                        existing.bbox.y1 = Math.max(existing.bbox.y1, w.bbox.y1);
                        existing.bbox.x0 = Math.min(existing.bbox.x0, w.bbox.x0);
                        existing.bbox.y0 = Math.min(existing.bbox.y0, w.bbox.y0);
                    } else {
                        idrNumbers.push({...w, value: val});
                    }
                }
            });

            idrNumbers.forEach(row => {
                let val100 = row.value; 
                let val80 = Math.round(val100 * 0.8);
                let val20 = val100 - val80;

                let words70 = words.filter(w => Math.abs(w.centerY - row.centerY) < 15 && Math.abs(w.centerX - header70.centerX) < 70 && cleanNumber(w.text) > 0);
                if (words70.length > 0) {
                    patchText(words70, formatRupiah(val80), false);
                }

                let words30 = words.filter(w => Math.abs(w.centerY - row.centerY) < 15 && Math.abs(w.centerX - header30.centerX) < 70 && cleanNumber(w.text) > 0);
                if (words30.length > 0) {
                    patchText(words30, formatRupiah(val20), false);
                }
            });

            ctx2.save();
            let wmSize = Math.max(30, imgElement.height * 0.035);
            ctx2.font = `900 ${wmSize}px 'Arial Black', 'Segoe UI Black', sans-serif`;
            ctx2.fillStyle = "rgba(100, 100, 100, 0.85)";
            ctx2.textAlign = "right";
            ctx2.textBaseline = "bottom";
            ctx2.fillText("TNC7 MEMBER", canvas2.width - (wmSize), canvas2.height - (wmSize));
            ctx2.restore();

            log("<span class='text-emerald-400 font-bold'>SUKSES: Perfect Clone V5 Selesai! Silakan simpan.</span>");
            document.getElementById('scanLine').style.display = 'none';
            document.getElementById('outputPrompt').classList.add('hidden');
            document.getElementById('dropZone').classList.add('hidden');
            canvas2.classList.remove('hidden');
            
            document.getElementById('downloadContainer').classList.remove('hidden');
            document.getElementById('btnDownloadXlsx').classList.add('hidden');
        }
